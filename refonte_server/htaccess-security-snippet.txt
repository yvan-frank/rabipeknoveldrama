# Directives de défense en profondeur à AJOUTER au .htaccess existant du
# dossier de l'app sur LWS — ne remplace PAS le fichier généré par cPanel
# (Setup Node.js App), qui contient les directives Passenger indispensables
# au fonctionnement de l'app (PassengerAppRoot / PassengerBaseURI). Copie
# juste ce qui suit à la fin de ce .htaccess-là.
#
# But : si jamais Apache sert aussi des fichiers statiques dans ce même
# dossier (au lieu de tout faire passer par Passenger), ces fichiers ne
# doivent jamais être accessibles publiquement.

# Fichiers de configuration/secrets
<FilesMatch "^\.env">
  Require all denied
</FilesMatch>

# Sourcemaps (révèlent le code TypeScript source) et fichiers de config projet
<FilesMatch "\.(map|json|lock|yml|yaml)$">
  Require all denied
</FilesMatch>
# ... mais package.json seul n'a rien de secret : autoriser explicitement si besoin.
# <Files "package.json"> Require all granted </Files>

# Dossiers qui ne doivent jamais être servis tels quels
RewriteEngine On
RewriteRule ^(prisma|src|\.git|node_modules)/ - [F,L]
